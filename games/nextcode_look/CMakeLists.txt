# next_code's look is in its fonts: a build without font files
# (KIN_ENABLE_TTF off) leaves it out.
if(NOT KIN_ENABLE_TTF)
    return()
endif()

add_executable(nextcode_look
    src/main.cpp
)

target_link_libraries(nextcode_look PRIVATE
    kin::engine
)

kin_game(nextcode_look TITLE "next_code Look" NO_CONTENT)

target_compile_options(nextcode_look PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
    $<$<CXX_COMPILER_ID:MSVC>:/W4>
)

# next_code's icons (icons/*.svg), compiled in as string constants in
# nextcode_icons.hpp: icons::rust, icons::folder_open, ...
file(GLOB nextcode_icon_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/icons/*.svg")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${nextcode_icon_files})
set(nextcode_icons "#pragma once\n\n// Generated by games/nextcode_look/CMakeLists.txt from icons/*.svg.\n\n#include <string_view>\n\nnamespace icons {\n")
foreach(icon_file IN LISTS nextcode_icon_files)
    get_filename_component(icon_name "${icon_file}" NAME_WE)
    file(READ "${icon_file}" icon_svg)
    string(STRIP "${icon_svg}" icon_svg)
    string(APPEND nextcode_icons "inline constexpr std::string_view ${icon_name} = R\"svg(${icon_svg})svg\";\n")
endforeach()
string(APPEND nextcode_icons "} // namespace icons\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/nextcode_icons.hpp" CONTENT "${nextcode_icons}" @ONLY)
target_include_directories(nextcode_look PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

# Headless it lays the window out with the File menu open.
add_test(NAME nextcode-look
    COMMAND nextcode_look --headless --frames=5 --menu=File --report=-
)
